\begin{frame}{Analytic Setting}
  \begin{itemize}
    \item Space, norm, topology, measure, or operator algebra.
    \item Convergence mode: pointwise, norm, weak, weak-* or distributional.
    \item Standing hypotheses and constants that must persist.
    \item Ledger: source for definitions and conventions.
  \end{itemize}
\end{frame}

\begin{frame}{Main Estimate Or Principle}
  \begin{theorem}
    State the boundedness, compactness, representation, convergence, or
    regularity result with all hypotheses.
  \end{theorem}
  \begin{itemize}
    \item Source: theorem number, lecture note section, or standard reference.
    \item Status: proved theorem, conditional theorem, example, or counterexample.
    \item Ledger: record constants, spaces, and endpoint cases.
  \end{itemize}
\end{frame}

\begin{frame}{Proof Mechanism}
  \begin{enumerate}
    \item Approximation, density, duality, or interpolation step.
    \item Compactness, spectral, maximal, or contour argument.
    \item Limiting step and where hypotheses are used.
  \end{enumerate}
\end{frame}

\begin{frame}{Examples And Counterexamples}
  \begin{itemize}
    \item Example showing the theorem in its natural setting.
    \item Endpoint or non-example showing why an assumption is needed.
    \item Ledger: source each example separately from the theorem.
  \end{itemize}
\end{frame}
