\begin{frame}{Model}
  \begin{itemize}
    \item Source, channel, code, protocol, adversary, oracle, or leakage model.
    \item Alphabet, block length, rate, noise, key size, or security parameter.
    \item Asymptotic regime and probability convention.
    \item Ledger: source for model and threat assumptions.
  \end{itemize}
\end{frame}

\begin{frame}{Claim To Prove Or Measure}
  \begin{block}{Target}
    Capacity, achievability, converse, distance, decoding radius, indistinguishability, or reduction.
  \end{block}
  \begin{itemize}
    \item Separate theorem statements from empirical attack or decoding results.
    \item Mark public parameters and hidden assumptions.
  \end{itemize}
\end{frame}

\begin{frame}{Main Result}
  \begin{theorem}
    State the rate, capacity, error, security, or reduction guarantee.
  \end{theorem}
  \begin{itemize}
    \item Source: theorem number, construction, experiment, or standard bound.
    \item Status: proved, random-coding argument, reduction, or empirical result.
    \item Ledger: record adversary model, rate, probability, and constants.
  \end{itemize}
\end{frame}

\begin{frame}{Proof Or Reduction}
  \begin{enumerate}
    \item Encoding/decoding, entropy, typicality, hybrid, or game hop.
    \item Error or advantage bound.
    \item Converse, lower bound, or attack limitation.
  \end{enumerate}
\end{frame}
