\begin{frame}{Pseudocode}
  \begin{enumerate}
    \item Preprocess the input.
    \item Repeat until stopping condition:
      \begin{enumerate}
        \item choose candidate;
        \item update certificate;
        \item check failure case.
      \end{enumerate}
    \item Return output and certificate.
  \end{enumerate}
\end{frame}

\begin{frame}{Correctness Invariant}
  \begin{lemma}
    The maintained certificate implies the desired output property.
  \end{lemma}
\end{frame}
